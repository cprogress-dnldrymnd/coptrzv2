# This block does NOT belong in this git repository — it belongs in
# app/public/.htaccess (the WordPress site root), OUTSIDE wp-content/themes/coptrz
# entirely. That file has no version control of its own, so this is the only
# place this content is preserved. To deploy: paste this block into
# app/public/.htaccess, between the existing "# BEGIN WordPress" / "# END WordPress"
# markers and anything after them — order relative to the WordPress block doesn't
# matter, mod_rewrite and mod_headers are independent.
#
# This is a mirror of dd_send_security_headers() / coptrz_csp_directives() in
# wp-content/themes/coptrz/includes/hooks.php (branch: security-headers, at the
# 2026-07 peak of the security-headers hardening work, commit 0ef041ad). Needed
# because on a LiteSpeed full-page-cache HIT the PHP-emitted headers are bypassed
# entirely — this is the only path that guarantees the headers on a cached
# response. KEEP THIS IN SYNC with hooks.php manually if that policy ever changes
# — there is no shared source between the two.
#
# wp-admin/wp-login are excluded from the enforcing CSP (env=!COPTRZ_IS_ADMIN):
# the block editor needs eval()/blob: workers a script-src allowlist would break.

# BEGIN Coptrz Security Headers
<IfModule mod_setenvif.c>
SetEnvIf Request_URI "^/wp-admin/" COPTRZ_IS_ADMIN
SetEnvIf Request_URI "wp-login\.php" COPTRZ_IS_ADMIN
</IfModule>

<IfModule mod_headers.c>
Header always unset X-Powered-By
Header always set X-Content-Type-Options "nosniff"
Header always set X-Frame-Options "SAMEORIGIN"
Header always set Referrer-Policy "strict-origin-when-cross-origin"
Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), usb=(), serial=(), interest-cohort=(), payment=(self)"
Header always set Cross-Origin-Opener-Policy "same-origin-allow-popups"

Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' 'unsafe-eval' https://bzrcdn.openai.com https://www.youtube.com https://www.googletagmanager.com https://www.google-analytics.com https://static.hotjar.com https://script.hotjar.com https://*.hotjar.io https://widget.intercom.io https://js.intercomcdn.com https://*.booqable.com https://*.booqableshop.com https://admin.revenuehunt.com https://www.google.com https://www.gstatic.com https://*.doubleclick.net https://www.googleadservices.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com https://*.booqable.com https://*.booqableshop.com; font-src 'self' data: https://fonts.gstatic.com https://*.booqable.com https://*.booqableshop.com; img-src 'self' data: https:; media-src 'self' data: blob: https:; connect-src 'self' https://*.hotjar.com wss://*.hotjar.com https://*.hotjar.io wss://*.hotjar.io https://api-iam.intercom.io https://*.booqable.com https://*.booqableshop.com https://www.google.com https://*.doubleclick.net https://www.googleadservices.com https://bzr.openai.com https://bzrcdn.openai.com; frame-src 'self' https://www.youtube.com https://*.booqable.com https://*.booqableshop.com https://www.google.com; worker-src 'self' blob:; object-src 'none'; base-uri 'self'; form-action 'self' https://shop.coptrz.com; frame-ancestors 'self'; upgrade-insecure-requests" env=!COPTRZ_IS_ADMIN
Header always set Content-Security-Policy "frame-ancestors 'self'" env=COPTRZ_IS_ADMIN

# HTTPS only, mirrors the is_ssl() check in dd_send_security_headers().
Header always set Strict-Transport-Security "max-age=31536000" env=HTTPS
</IfModule>
# END Coptrz Security Headers
